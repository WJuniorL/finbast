function ExternalAction({ open, aberto, onOpenChange, aoMudarAbertura, onClose, titulo }) {
  useEffect(() => {
    if (open === false || aberto === false) return;
    signal(titulo ? `Dossiê: ${titulo}. Abertura simulada nesta prévia.` : 'Ação da ficha original. Formulário completo disponível no sistema.');
    onOpenChange?.(false); aoMudarAbertura?.(false); onClose?.();
  }, [open, aberto]);
  return null;
}
